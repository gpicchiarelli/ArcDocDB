(defparameter *checks* nil)
(defparameter *safe-reads* 0)
(defparameter *byte-comparisons* nil)
(defun ensure-audit (test label &optional detail)
  (unless test (error "Evidence audit: ~S ~S" label detail))
  (push (list :check label :status :ok :detail detail) *checks*) t)
(defun bytes (path)
  (with-open-file (s path :element-type '(unsigned-byte 8))
    (let ((v (make-array (file-length s) :element-type '(unsigned-byte 8))))
      (assert (= (read-sequence v s) (length v)))
      (assert (eq (read-byte s nil :eof) :eof)) v)))
(defun md5-bytes (v)
  (format nil "~(~{~2,'0X~}~)" (coerce (sb-md5:md5sum-sequence v) 'list)))
(defun text (v) (sb-ext:octets-to-string v :external-format :utf-8))
(defun safe-string (string)
  (let ((*read-eval* nil) (eof (gensym)))
    (with-input-from-string (s string)
      (let ((d (read s nil eof)))
        (assert (not (eq d eof)))
        (assert (eq (read s nil eof) eof))
        (incf *safe-reads*) d))))
(defun data (path) (safe-string (text (bytes path))))
(defun schema (d) (or (getf d :schema-version) (getf d :schema)))
(defun local (name) (merge-pathnames name *base*))
(defun source (name) (merge-pathnames name *root*))
(defun provided (d key)
  (let ((missing (gensym))) (not (eq (getf d key missing) missing))))
(defun same-bytes (original copy role)
  (let ((a (bytes original)) (b (bytes copy)))
    (assert (equalp a b))
    (push (list :role role :source (namestring original)
                :copy (enough-namestring copy *base*) :bytes (length a)
                :md5 (md5-bytes a)) *byte-comparisons*) t))
(defun tree-files (dir)
  (append (uiop:directory-files dir)
          (loop for sub in (uiop:subdirectories dir) append (tree-files sub))))
(defun metadata-match (entry d)
  (assert (provided entry :formats)) (assert (null (getf entry :formats)))
  (dolist (pair '((:artifact-schema-version . :schema-version)
                  (:artifact-kind . :kind) (:status . :status)
                  (:source-consistency . :source-consistency) (:exit-code . :exit-code)))
    (when (provided entry (car pair))
      (assert (equalp (getf entry (car pair))
                      (if (eq (car pair) :artifact-schema-version) (schema d)
                          (getf d (cdr pair))))))) t)
