(defun ca-read-one (path)
  (let ((*read-eval* nil))
    (with-open-file (stream path)
      (let ((value (read stream nil :eof)))
        (when (eq value :eof) (error "Empty data file."))
        (unless (eq (read stream nil :eof) :eof) (error "Trailing data."))
        value))))
