(require :asdf)
(require :sb-md5)
(defun audit-check (truth message)
  (unless truth (error "Audit CBOR: ~A" message)))
(defun safe-one (stream label)
  (let ((*read-eval* nil))
    (let ((data (read stream nil :eof)))
      (audit-check (and (consp data) (eq :eof (read stream nil :eof)))
                   (format nil "single-form/EOF ~A" label))
      data)))
(defun safe-file (path)
  (with-open-file (s path :external-format :utf-8) (safe-one s path)))
(defun safe-string (text)
  (audit-check (stringp text) "stdout non testuale")
  (with-input-from-string (s text) (safe-one s "stdout")))
(defun byte-vector (path)
  (with-open-file (s path :element-type '(unsigned-byte 8))
    (let* ((size (file-length s)) (bytes (make-array size :element-type '(unsigned-byte 8))))
      (audit-check (= size (read-sequence bytes s)) "lettura completa bytevector")
      bytes)))
(defun digest (path)
  (format nil "~(~{~2,'0X~}~)" (coerce (sb-md5:md5sum-file path) 'list)))
(defun safe-relative (path)
  (audit-check (and (stringp path) (plusp (length path))
                    (not (uiop:absolute-pathname-p path))
                    (not (member :up (pathname-directory path)))
                    (not (member ".." (pathname-directory path) :test #'equal)))
               "percorso fingerprint non locale")
  path)
(defun fingerprint-audit (data)
  (let ((before (getf data :source-fingerprints-before))
        (after (getf data :source-fingerprints-after)))
    (audit-check (and before (equal before after) (eq :stable (getf data :source-consistency)))
                 "fingerprint prima/dopo non STABLE")
    (audit-check (= (length before)
                    (length (remove-duplicates before :key (lambda (x) (getf x :file)) :test #'equal)))
                 "fingerprint duplicati")
    (dolist (entry before)
      (audit-check (equal (getf entry :md5) (digest (safe-relative (getf entry :file))))
                   (format nil "fingerprint corrente ~A" (getf entry :file))))
    (list :before-after-current-match t :count (length before))))
(defun wrapper-audit (path expected-kind)
  (let* ((wrapper (safe-file path)) (data (safe-string (getf wrapper :stdout)))
         (before (getf wrapper :source-blobs-before)) (after (getf wrapper :source-blobs-after)))
    (audit-check (and (= 1 (getf wrapper :schema-version))
                      (eq :command-verification (getf wrapper :kind))
                      (eq :ok (getf wrapper :status)) (eql 0 (getf wrapper :exit-code))
                      (eq :stable (getf wrapper :source-consistency))
                      before (equal before after))
                 (format nil "wrapper non PASS/STABLE ~A" path))
    (audit-check (and (= 1 (getf data :schema-version)) (eq expected-kind (getf data :kind))
                      (eq :ok (getf data :status))) "stdout non PASS/schema1/kind atteso")
    (let* ((files (mapcar (lambda (entry) (safe-relative (getf entry :path))) before))
           (hashes (uiop:split-string
                    (uiop:run-program (append '("git" "hash-object" "--") files) :output :string)
                    :separator '(#\Newline)))
           (actual (remove "" hashes :test #'string=)))
      (audit-check (= (length before) (length actual)) "numero blob correnti")
      (let ((nonfocus-changes nil))
        (loop for entry in before for blob in actual
              for file = (getf entry :path)
              unless (equal blob (getf entry :git-blob))
                do (audit-check (and (not (string= file "arcdocdb.asd"))
                                     (not (uiop:string-prefix-p "src/" file))
                                     (not (uiop:string-prefix-p "tests/" file))
                                     (not (uiop:string-prefix-p "tools/" file)))
                                (format nil "blob corrente prodotto/test/tool/ASDF ~A" file))
                   (push (list :path file :at-campaign (getf entry :git-blob) :current blob)
                         nonfocus-changes))
        (values data (list :path path :wrapper-md5 (digest path) :protected-single-form-stdout t
                          :source-blobs-before-after-match t :blob-count (length before)
                          :current-src-tests-tools-asdf-match t
                          :post-campaign-nonfocus-changes (nreverse nonfocus-changes)
                          :product-fingerprints (fingerprint-audit data)))))))
(defun lines (text) (uiop:split-string text :separator '(#\Newline)))
(defun exact-line (text expected)
  (find expected (lines text) :test #'string= :key (lambda (line) (string-right-trim '(#\Return) line))))
(defun compile-failure (text)
  (some (lambda (marker) (search marker text :test #'char-equal))
        '("compilation aborted" "COMPILE-FILE-ERROR" "COMPILE-FILE-WARNED")))
(defun expected-sink (iterations token)
  (logand most-positive-fixnum (+ (* iterations token) (floor (* iterations (1- iterations)) 2))))
(defun sample-audit (sample iterations warmup token units &key positive)
  (audit-check (and (= iterations (getf sample :iterations))
                    (= warmup (getf sample :warmup-iterations))
                    (= token (getf sample :expected-return-token))
                    (= (expected-sink iterations token) (getf sample :sink) (getf sample :expected-sink))
                    (typep (getf sample :raw-ticks) '(integer 0))
                    (typep (getf sample :heap-bytes) '(integer 0))
                    (= (getf sample :seconds) (/ (getf sample :raw-ticks) (float units 1d0))))
                 "iterazioni/warmup/token/sink/timer incoerenti")
  (audit-check (if positive (>= (getf sample :heap-bytes) (* 16 1048576))
                         (zerop (getf sample :heap-bytes))) "heap sample")
  (audit-check (if (plusp (getf sample :seconds))
                   (= (getf sample :calls-per-second) (/ iterations (getf sample :seconds)))
                   (null (getf sample :calls-per-second))) "derivazione calls/sec")
  t)
(defun sensor-audit (data)
  (let ((self (getf data :self-test)) (units (getf data :timer-units-per-second)))
    (audit-check (and (eq :ok (getf self :status)) (typep units '(integer 1))
                      (= 1 (getf data :workers)) (= 3 (getf data :safety))) "self sensor/meta")
    (sample-audit (getf self :baseline) 32 128 0 units)
    (sample-audit (getf self :positive-control) 16 0 1048576 units :positive t)
    (list :baseline-heap-bytes 0 :positive-heap-bytes (getf (getf self :positive-control) :heap-bytes)
          :positive-expected-minimum (* 16 1048576) :units-per-second units)))
(defun benchmark-audit (data)
  (let ((cells (getf data :campaigns)) (summary nil)
        (definitions '((:scalar-u64 9 (1 0 11)) (:ascii-32k 32771 (1 0 32773))
                       (:arrays-100 100 (100 100 102)) (:mixed 21 (12 2 23))
                       (:array-4096 4099 (4097 1 4101)) (:tags-4096 4097 (4097 0 4099)))))
    (audit-check (equal (mapcar (lambda (cell) (getf cell :name)) cells)
                        (mapcar #'first definitions)) "nomi/ordine sei fixture")
    (loop for cell in cells for (name size expected) in definitions
          do (let* ((token (+ (first expected) (* 65536 (second expected)) (* 16777216 (third expected))))
                    (samples (getf cell :samples)))
               (audit-check (and (= size (getf cell :unit-bytes)) (= 2 (getf cell :start))
                                 (= (+ size 2) (getf cell :end))
                                 (equal expected (getf cell :expected-values))
                                 (= token (getf cell :expected-token))
                                 (eq t (getf cell :input-unchanged))
                                 (eq t (getf cell :private-preallocated-space)) (= 5 (length samples)))
                            "fixture/tre valori/scratch/immutabilita/repliche")
               (dolist (sample samples)
                 (sample-audit sample 32 128 token (getf data :timer-units-per-second)))
               (push (list :name name :bytes size :expected-three-values expected :token token
                           :samples 5 :heap-bytes '(0 0 0 0 0)
                           :raw-ticks (mapcar (lambda (x) (getf x :raw-ticks)) samples)
                           :verified-independent-sink (expected-sink 32 token)) summary)))
    (list :status :ok :fixtures 6 :samples 30 :measured-calls 960 :warmup-calls 3840
          :all-three-values-and-input-immutability-checked t
          :sensor (sensor-audit data) :cells (nreverse summary))))

    ("utf8-major" "GENERATED-AST-CORPUS" "UTF8-LEADING all'offset 22" "Bytes validi arbitrari vengono sottoposti erroneamente a UTF8.")
    ("map-child-arity" "RFC-DEFINITE-VECTORS" "CBOR-TRUNCATED all'offset 8" "Mappa definitiva vuota richiede erroneamente due figli.")))
(defun text (path) (uiop:read-file-string path :external-format :utf-8))
(defun copy-source-paths ()
  (append '(#p"arcdocdb.asd" #p"tools/build.lisp")
          (sort (append (directory "src/**/*.lisp") (directory "tests/**/*.lisp")) #'string< :key #'namestring)))
(defun replace-target (body before after)
  (let ((pos (search before body)))
    (audit-check (and pos (not (search before body :start2 (1+ pos)))) "target non unico")
    (concatenate 'string (subseq body 0 pos) after (subseq body (+ pos (length before))))))
(defun mutation-audit (data)
  (let* ((base (getf data :baseline)) (base-log (getf base :log)) (base-text (text base-log))
         (mutants (getf data :mutants)) (targets (getf data :targets))
         (sources (copy-source-paths)) (summary nil))
    (audit-check (and (eq :ok (getf base :status)) (= 0 (getf base :exit-code))
                      (not (compile-failure base-text))
                      (exact-line base-text "ok    ARCDOCDB:*VERSION* è una stringa")
                      (exact-line base-text "build e test: nessun avviso, tutti i controlli superati"))
                 "baseline completa/esatta/compilabile")
    (audit-check (= 320 (count-if (lambda (line) (uiop:string-prefix-p "ok    " line)) (lines base-text)))
                 "conteggio baseline ASDF320 incl2smoke")
    (dolist (source sources)
      (audit-check (equalp (byte-vector source)
                           (byte-vector (merge-pathnames (enough-namestring source)
                                                         "spikes/out/cbor-structure-mutations/baseline/")))
                   "baseline copia corrente non identica"))
    (audit-check (and (= 10 (length mutants) (length targets))
                      (equal (mapcar (lambda (x) (getf x :name)) mutants) (mapcar #'first *causes*)))
                 "dieci mutanti e nomi/ordine")
    (loop for mutant in mutants for target in targets for cause in *causes* for i from 0
          do (let* ((log (getf mutant :log)) (body (text log)) (rows (lines body))
                    (first-error (position-if (lambda (line) (uiop:string-prefix-p "Unhandled " line)) rows))
                    (header (search "ok    ARCDOCDB:*VERSION* è una stringa" body))
                    (failure (search "Unhandled " body)) (path (getf target :source-file))
                    (changed nil))
               (audit-check (and (eq :detected (getf mutant :result)) (eq t (getf mutant :detected))
                                 (eql 1 (getf mutant :exit-code)) (null (getf mutant :diagnostic))
                                 (equal path (getf mutant :source-file))
                                 (exact-line body "ok    ARCDOCDB:*VERSION* è una stringa")
                                 (not (compile-failure body)) first-error header failure (< header failure)
                                 (search "ARCDOCDB.CBOR.STRUCTURE.TESTS" body)
                                 (search (second cause) body) (search (third cause) body)
                                 (not (exact-line body "build e test: nessun avviso, tutti i controlli superati")))
                            "mutante non compilato/rilevato dopo smoke o causa inattesa")
               (dolist (source sources)
                 (let* ((relative (enough-namestring source))
                        (copy (merge-pathnames relative (format nil "spikes/out/cbor-structure-mutations/~D/" i)))
                        (original (text source)) (copied (text copy)))
                   (unless (equal original copied) (push relative changed))
                   (audit-check (equal copied (if (equal relative path)
                                                   (replace-target original (getf target :before) (getf target :after))
                                                   original)) "copia mutata diversa dal singolo target")))
               (audit-check (equal changed (list path)) "non un solo source mutato")
               (let ((fasl (merge-pathnames (make-pathname :type "fasl" :defaults path)
                                           (format nil "spikes/out/cbor-structure-mutations/~D/fasl/" i))))
                 (audit-check (probe-file fasl) "FASL bersaglio assente")
                 (push (list :name (first cause) :result :detected :exit-code 1
                             :compiled-target-fasl (namestring fasl) :isolated-only-target-difference t
                             :log log :log-md5 (digest log) :first-kill (fourth cause)
                             :cause-lines (subseq rows first-error (min (length rows) (+ first-error 15)))) summary))))
    (list :status :ok :detected 10 :invalid 0 :survived 0 :baseline-exact-completion t
          :baseline-asdf-tests-including-two-smoke 320 :cbor-structure-tests 24
          :baseline-copy-files (length sources) :each-copy-only-single-target-change t
          :cases (nreverse summary))))
(let ((report (list :schema-version 1 :kind :cbor-structure-measurement-audit :status :running
                    :formats nil :recorded-at (get-universal-time) :reader-evaluation nil
                    :source-and-drivers-read-only t :product-or-campaign-reruns nil
                    :audit-repair-notes
                    '(:temporary-reader-unavailable-uiop-symbol-replaced-with-byte-vector-reader
                      :global-current-wrapper-comparison-refined-to-src-tests-tools-asdf
                      :post-campaign-document-changes-listed-separately)
                    :prior-failed-audit "/tmp/cbor-structure-measurement-audit-v1-global-current-assumption.lisp"
                    :limits '(:serial-success-path-samples :sensor-not-absolute-allocation-proof
                              :no-scaling-or-throughput-threshold :targeted-ten-mutants-only
                              :first-tag-partition-kill-from-simple-false :no-full-c1-mcdc-or-engine-approval))))
  (handler-case
      (multiple-value-bind (self self-wrapper)
          (wrapper-audit "spikes/out/4000529099-command-82288-0/report.lisp" :cbor-structure-benchmark)
        (multiple-value-bind (bench bench-wrapper)
            (wrapper-audit "spikes/out/4000529113-command-82449-0/report.lisp" :cbor-structure-benchmark)
          (multiple-value-bind (mut-self mut-self-wrapper)
              (wrapper-audit "spikes/out/4000529069-command-81915-0/report.lisp" :cbor-structure-mutations)
            (multiple-value-bind (mut mut-wrapper)
                (wrapper-audit "spikes/out/4000529135-command-82763-0/report.lisp" :cbor-structure-mutations)
              (audit-check (equal mut (safe-file "spikes/out/cbor-structure-mutations/report.lisp"))
                           "raw mutazioni diverso da stdout wrapper")
              (audit-check (and (eq :ok (getf (getf mut-self :self-test) :status))
                                (= 10 (getf (getf mut-self :self-test) :applicable-mutants))
                                (eq :invalid (getf (getf mut-self :self-test) :missing-baseline-completion)))
                           "self-test mutator metadata")
              (setf (getf report :benchmark-self-test) (sensor-audit self)
                    (getf report :benchmark) (benchmark-audit bench)
                    (getf report :mutations) (mutation-audit mut)
                    (getf report :wrapper-audits) (list self-wrapper bench-wrapper mut-self-wrapper mut-wrapper)
                    (getf report :status) :ok)))))
    (error (condition) (setf (getf report :status) :failed (getf report :diagnostic) (princ-to-string condition))))
  (with-open-file (s "/tmp/cbor-structure-measurement-audit.lisp" :direction :output
                     :if-exists :error :if-does-not-exist :create :external-format :utf-8)
    (let ((*print-readably* t)) (write report :stream s :pretty t) (terpri s)))
  (write (list :status (getf report :status) :diagnostic (getf report :diagnostic)
               :benchmark-samples (getf (getf report :benchmark) :samples)
               :mutation-detected (getf (getf report :mutations) :detected)))
  (terpri)
  (unless (eq :ok (getf report :status)) (sb-ext:exit :code 1)))
