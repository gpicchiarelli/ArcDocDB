(defun read-datum (stream)
  (let ((*read-eval* nil) (*package* (find-package :cl-user)) (eof (gensym)))
    (let ((data (read stream nil eof)))
      (when (eq data eof) (error "Datum vuoto"))
      (unless (eq eof (read stream nil eof)) (error "Datum oltre EOF")) data)))
(defun data-file (path) (with-open-file (s path :external-format :utf-8) (read-datum s)))
(defun data-string (text) (with-input-from-string (s text) (read-datum s)))
(defun bytes-file (path)
  (with-open-file (s path :element-type '(unsigned-byte 8))
    (let ((bytes (make-array (file-length s) :element-type '(unsigned-byte 8))))
      (unless (= (length bytes) (read-sequence bytes s)) (error "Short read ~A" path))
      (unless (eq :eof (read-byte s nil :eof)) (error "Beyond EOF ~A" path)) bytes)))

(defun weighted-token (values)
  (destructuring-bind (major ai high low next form) values
    (+ major (* 8 ai) (* 256 high) (* 65536 low) (* 1048576 next)
       (* 16777216 (ecase form (:argument 1) (:indefinite 2) (:break 3))))))

(defun check-sample (sample n warmup token timer)
  (let ((sink (logand most-positive-fixnum (+ (* n token) (/ (* n (1- n)) 2))))
        (ticks (getf sample :raw-ticks)))
    (assure (and (= n (getf sample :iterations)) (= warmup (getf sample :warmup-iterations))
                 (= token (getf sample :expected-return-token))
                 (= sink (getf sample :sink) (getf sample :expected-sink))) "Sink/calls mismatch")
    (assure (and (integerp ticks) (not (minusp ticks))
                 (close-number (/ ticks (coerce timer 'double-float)) (getf sample :seconds))) "Timer mismatch")
    (if (zerop ticks) (assure (null (getf sample :calls-per-second)) "Zero time rate")
        (assure (close-number (/ n (getf sample :seconds)) (getf sample :calls-per-second)) "Rate mismatch"))))
