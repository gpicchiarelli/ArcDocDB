;;;; Collector di dati; nessun sorgente del prodotto viene caricato o valutato.
;;;; Uso: sbcl --noinform --no-userinit --no-sysinit --script QUESTO-FILE CONFIG.lisp
(require :asdf)
(require :sb-posix)
(defpackage #:writer-queue.evidence (:use #:cl))
(in-package #:writer-queue.evidence)
(declaim (optimize (safety 3)))
(defvar *root*)
(defvar *jobs* nil)
(defvar *entries* nil)
(defvar *raw-entries* nil)
(defvar *reports* nil)

(defun read-data-stream (stream)
  "Legge esattamente un dato, con read-eval disattivato e guardia EOF."
  (let* ((*read-eval* nil) (eof (gensym "EOF")) (data (read stream nil eof)))
    (when (eq data eof) (error "Dati vuoti."))
    (unless (eq (read stream nil eof) eof) (error "Dati aggiuntivi dopo il dato.")) data))
(defun read-data (path)
  (with-open-file (stream path :external-format :utf-8) (read-data-stream stream)))
(defun read-byte-datum (bytes)
  (with-input-from-string (stream (sb-ext:octets-to-string bytes :external-format :utf-8))
    (read-data-stream stream)))
(defun source-path (value)
  (truename (merge-pathnames value *root*)))
(defun read-bytes (path)
  "Acquisisce i byte prima di creare qualsiasi target."
  (with-open-file (stream path :element-type '(unsigned-byte 8))
    (let ((bytes (make-array (file-length stream) :element-type '(unsigned-byte 8))))
      (unless (= (read-sequence bytes stream) (length bytes)) (error "Dati accorciati: ~A" path))
      (unless (eq (read-byte stream nil :eof) :eof) (error "Dati cresciuti: ~A" path)) bytes)))
(defun read-text (path)
  (sb-ext:octets-to-string (read-bytes path) :external-format :utf-8))
(defun data-bytes (value)
  (sb-ext:string-to-octets
   (with-output-to-string (stream)
     (let ((*print-readably* t) (*print-pretty* t) (*print-length* nil) (*print-level* nil))
       (write value :stream stream) (terpri stream))) :external-format :utf-8))
(defun safe-local (name)
  "Richiede un nome relativo senza traversamento, componenti wildcard o NUL."
  (unless (and (stringp name) (plusp (length name))) (error "Target mancante."))
  (let ((path (pathname name)))
    (unless (and (stringp name) (plusp (length name))
                 (not (find #\Null name)) (not (find #\\ name))
                 (not (wild-pathname-p path))
                 (not (member :absolute (pathname-directory path)))
                 (not (member :up (pathname-directory path)))
                 (not (member :back (pathname-directory path)))
                 (not (search "../" name)) (not (string= name "..")))
      (error "Nom relatif invalido: ~S" name))) name)
(defun raw-prefix (artifact suffix)
  (let ((name (pathname artifact)))
    (concatenate 'string (or (pathname-name name) (error "Artifact senza nome.")) suffix)))
(defun schema-value (data)
  (or (getf data :schema-version) (getf data :schema)))
(defun schema-one (data)
  (unless (and (listp data) (eql (schema-value data) 1))
    (error "Schema dati non supportato.")) data)
(defun push-job (local bytes &key source data (operation :write))
  (safe-local local)
  (when (find local *jobs* :key (lambda (job) (getf job :local)) :test #'string=)
    (error "Target duplicato nel piano: ~A" local))
  (push (list :local local :bytes bytes :source source :data data :operation operation) *jobs*) local)
(defun byte-job (source local &optional data)
  (let ((path (source-path source)))
    (push-job local (read-bytes path) :source (namestring path) :data data)))
(defun datum-job (local data)
  (push-job local (data-bytes data) :data data))
(defun entry (kind artifact &rest metadata)
  (let ((record (append (list :kind kind :formats nil :artifact artifact) metadata)))
    (if (or (find #\/ artifact) (not (string-equal (or (pathname-type (pathname artifact)) "") "lisp")))
        (push record *raw-entries*)
        (push record *entries*))))
(defun report-metadata (data)
  (list :artifact-schema-version (schema-value data) :artifact-kind (getf data :kind)
        :status (getf data :status) :source-consistency (getf data :source-consistency)
        :exit-code (getf data :exit-code) :started-at (getf data :started-at-universal-time)
        :finished-at (getf data :finished-at-universal-time) :wall-seconds (getf data :wall-seconds)))
(defun report-by-artifact (artifact)
  (or (cdr (assoc artifact *reports* :test #'string=))
      (error "Wrapper non dichiarato: ~A" artifact)))

(defun collect-process (item)
  "Non scarta tentativi falliti; byte originali e stdout/stderr sono sempre conservati."
  (let* ((source (source-path (getf item :source))) (artifact (getf item :artifact))
         (data (schema-one (read-data source)))
         (prefix (raw-prefix artifact "")))
    (unless (eq (getf data :kind) :command-verification) (error "Wrapper inatteso: ~A" source))
    (byte-job source artifact data)
    (push (cons artifact data) *reports*)
    (apply #'entry (getf item :role) artifact (report-metadata data))
    (dolist (channel '(:stdout :stderr))
      (let* ((text (getf data channel)) (local (format nil "~A.~(~A~).log" prefix channel))
             (physical (getf item (if (eq channel :stdout) :stdout-source :stderr-source))))
        (when (and text (not (stringp text))) (error "Canale non stringa/NIL: ~A ~S" source channel))
        (cond
          ((and (null text) (null physical))
           (entry :raw-process-output-absence artifact :process-artifact artifact :channel channel
                  :source-kind :wrapper-absent-channel :embedded-value nil
                  :present-p (not (null (nth-value 2 (get-properties data (list channel)))))))
          (t
        (if physical
            (progn
              (when (and text (not (string= text (read-text (source-path physical)))))
                (error "Canale fisico diverso dal wrapper: ~A" physical))
              (byte-job physical local))
            (push-job local (sb-ext:string-to-octets text :external-format :utf-8)))
        (entry :raw-process-output local :process-artifact artifact :channel channel
               :source-kind (if physical :physical-byte-stream :wrapper-embedded-string)
               :encoding :utf-8 :characters (and text (length text)))))))
    (when (getf item :embedded-report-artifact)
      (let ((raw (handler-case
                     (with-input-from-string (stream (getf data :stdout))
                       (list :status :single-datum :data (read-data-stream stream)))
                   (error (condition) (list :status :unparsed :diagnostic (princ-to-string condition))))))
        (let ((local (getf item :embedded-report-artifact)))
          (datum-job local (list :schema-version 1 :kind :raw-process-datum :formats nil
                                :process-artifact artifact :parse-result raw :raw-stdout (getf data :stdout)))
          (entry :raw-process-datum local :process-artifact artifact))))))

(defun collect-coverage (item)
  "Preserva stato, ogni HTML e report completo; nessuna eccezione al denominatore."
  (let* ((artifact (getf item :artifact)) (process (getf item :process-artifact))
         (wrapper (report-by-artifact process))
         (base (source-path (uiop:ensure-directory-pathname (getf item :directory))))
         (prefix (raw-prefix artifact "-raw/"))
         (state-path (merge-pathnames "coverage-state.lisp" base))
         (state-text (read-text state-path))
         (index-name (or (getf item :index-name) "cover-index.html"))
         (htmls (sort (directory (merge-pathnames "*.html" base)) #'string< :key #'namestring))
         (html-data nil) (report-source (getf item :report-source)))
    (with-input-from-string (stream state-text) (read-data-stream stream))
    (unless (find index-name htmls :key #'file-namestring :test #'string=)
      (error "Indice HTML assente: ~A" index-name))
    (byte-job state-path (concatenate 'string prefix "coverage-state.lisp"))
    (entry :raw-coverage-state (concatenate 'string prefix "coverage-state.lisp") :process-artifact process)
    (dolist (path htmls)
      (let ((local (concatenate 'string prefix (file-namestring path))))
        (byte-job path local)
        (push (list :name (file-namestring path) :source-path (namestring path)
                    :local-file local :content (read-text path)) html-data)
        (entry :raw-coverage-html local :process-artifact process)))
    (let ((raw (list :schema-version 1 :kind :raw-coverage :formats nil
                     :scope '("src/codec/cbor-package.lisp" "src/codec/cbor-header.lisp")
                     :path-base "src/codec/" :test-scope "tests/codec/cbor-*.lisp"
                     :attempt (getf item :name) :process-artifact process :raw-process-report wrapper
                     :raw-sb-cover-state state-text :raw-state-source (namestring state-path)
                     :raw-state-local-file (concatenate 'string prefix "coverage-state.lisp")
                     :raw-index-html (read-text (merge-pathnames index-name base))
                     :raw-index-local-file (concatenate 'string prefix index-name)
                     :raw-html (nreverse html-data)
                     :limits '(:complete-raw-denominator :no-approved-exceptions :not-mcdc
                               :runtime-only :not-engine-or-release-qualification))))
      (when report-source
        (let* ((path (source-path report-source)) (data (read-data path))
               (local (concatenate 'string prefix "raw-report.lisp")))
          (byte-job path local data)
          (entry :raw-coverage-report local :process-artifact process)
          (setf (getf raw :raw-report) data (getf raw :raw-report-text) (read-text path)
                (getf raw :raw-report-source) (namestring path))))
      (datum-job artifact raw)
      (entry :raw-coverage artifact :artifact-schema-version 1 :artifact-kind :raw-coverage
             :process-artifact process :html-count (length htmls)))))

(defun declared-log-names (data)
  "Associa esclusivamente i nomi dichiarati dal report; non deduce esiti dai log."
  (let ((result nil) (baseline (getf data :baseline)))
    (when (getf baseline :log)
      (push (cons (namestring (source-path (getf baseline :log))) "baseline") result))
    (dolist (mutant (getf data :mutants))
      (when (getf mutant :log)
        (push (cons (namestring (source-path (getf mutant :log))) (getf mutant :name)) result))) result))
(defun collect-mutation (item)
  "Preserva l'intero report e tutti i .log, dichiarati o presenti nella directory."
  (let* ((artifact (getf item :artifact)) (process (getf item :process-artifact))
         (wrapper (report-by-artifact process)) (source (source-path (getf item :report-source)))
         (base (uiop:pathname-directory-pathname source)) (data (schema-one (read-data source)))
         (prefix (raw-prefix artifact "-logs/")) (declared (declared-log-names data))
         (paths (sort (remove-duplicates
                       (append (mapcar (lambda (pair) (pathname (car pair))) declared)
                               (directory (merge-pathnames "**/*.log" base)))
                       :test #'equal :key #'namestring) #'string< :key #'namestring))
         (logs nil))
    (byte-job source (raw-prefix artifact "-report-originale.lisp") data)
    (entry :raw-mutation-report (raw-prefix artifact "-report-originale.lisp") :process-artifact process)
    (loop for path in paths for n from 0
          do (let* ((relative (enough-namestring path base))
                    (inside (and (not (search "../" relative))
                                 (not (member :absolute (pathname-directory (pathname relative))))))
                    (local (concatenate 'string prefix
                                        (if inside relative (format nil "external-~D.log" n))))
                    (name (cdr (assoc (namestring path) declared :test #'string=))))
               (byte-job path local)
               (push (list :name name :source-path (namestring path) :local-file local
                           :content (read-text path)) logs)
               (entry :raw-mutation-log local :process-artifact process :name name)))
    (datum-job artifact (list :schema-version 1 :kind :raw-mutation-output :formats nil
                             :attempt (getf item :name) :process-artifact process :raw-process-report wrapper
                             :source-path (namestring source) :status (getf data :status)
                             :source-consistency (getf data :source-consistency)
                             :raw-report data :raw-report-text (read-text source) :logs (nreverse logs)
                             :limits (getf data :limits)))
    (entry :raw-mutation-output artifact :artifact-schema-version 1 :artifact-kind :raw-mutation-output
           :process-artifact process :status (getf data :status) :source-consistency (getf data :source-consistency)
           :log-count (length paths))))

(defun collect-review (item)
  "La lettura è attribuita alla fonte dichiarata, senza inferenza o approvazione umana."
  (let* ((artifact (getf item :artifact)) (data (copy-list (schema-one (getf item :data))))
         (provenance nil))
    (unless (stringp (getf data :statement-source)) (error "Provenienza della lettura assente."))
    (dolist (source-artifact (getf item :source-artifacts))
      (let ((report (report-by-artifact source-artifact)))
        (push (list :artifact source-artifact :status (getf report :status)
                    :source-consistency (getf report :source-consistency)
                    :source-blobs-before (getf report :source-blobs-before)
                    :source-blobs-after (getf report :source-blobs-after)
                    :source-fingerprints-before (getf report :source-fingerprints-before)
                    :source-fingerprints-after (getf report :source-fingerprints-after)) provenance)))
    (setf (getf data :formats) nil (getf data :source-provenance) (nreverse provenance))
    (datum-job artifact data)
    (entry :c1-review artifact :artifact-schema-version (schema-value data)
           :artifact-kind (getf data :kind) :status (getf data :status)
           :statement-source (getf data :statement-source))))

(defun collect-source-datum (item)
  "Copia bytewise review/audit esterni, mantenendo schema alias e provenienza originali."
  (let* ((source (source-path (getf item :source)))
         (data (schema-one (read-data source))) (artifact (getf item :artifact)))
    (byte-job source artifact data)
    (entry (getf item :role) artifact :artifact-schema-version (schema-value data)
           :artifact-schema-key (if (getf data :schema-version) :schema-version :schema)
           :artifact-kind (getf data :kind) :status (getf data :status)
           :statement-source (getf data :statement-source) :source-path (namestring source)
           :source-artifacts (getf item :source-artifacts))))
(defun collect-raw-file (item)
  (let ((source (source-path (getf item :source))) (artifact (getf item :artifact)))
    (byte-job source artifact)
    (entry (getf item :role) artifact :source-path (namestring source))))

(defun fresh-target (path)
  (when (probe-file path) (error "Target esistente conservato: ~A" path)) path)
(defun publish-appended-catalog (destination)
  "Rinomina il catalogo precedente in prior-catalog; stage pubblicato solo dopo verifica."
  (let* ((prior (find :rename *jobs* :key (lambda (job) (getf job :operation))))
         (stage (find :catalog-stage *jobs* :key (lambda (job) (getf job :operation))))
         (source (getf prior :source)) (history (merge-pathnames (getf prior :local) destination))
         (new (merge-pathnames (getf stage :local) destination)))
    (unless (and prior stage (equalp (read-bytes source) (getf prior :bytes)))
      (error "Catalogo precedente cambiato, dati conservati senza pubblicazione."))
    (fresh-target history)
    (rename-file source history)
    (handler-case
        (progn
          (fresh-target source)
          (unless (equalp (read-bytes history) (getf prior :bytes)) (error "Prior catalog incoerente."))
          (rename-file new source))
      (error (condition)
        (unless (probe-file source)
          (with-open-file (stream source :direction :output :element-type '(unsigned-byte 8)
                                        :if-exists :error :if-does-not-exist :create)
            (write-sequence (getf prior :bytes) stream)))
        (error condition)))))

(defun commit-plan (config)
  "Crea una directory esclusiva: ogni target usa IF-EXISTS ERROR e rilettura integrale."
  (let* ((relative (getf config :path-base))
         (destination (merge-pathnames (uiop:ensure-directory-pathname relative) *root*))
         (parent (make-pathname :defaults destination :directory (butlast (pathname-directory destination))
                                :name "segnaposto" :type nil))
         (append-p (eq (getf config :mode) :append)) (total 0))
    (safe-local relative)
    (if append-p
        (unless (uiop:directory-exists-p destination) (error "Cartella append assente."))
        (when (probe-file destination) (error "Cartella esistente conservata: ~A" destination)))
    (dolist (job *jobs*)
      (fresh-target (merge-pathnames (getf job :local) destination))
      (when (getf job :data)
        (unless (equalp (getf job :data) (read-byte-datum (getf job :bytes)))
          (error "Metadati diversi dai byte catturati: ~A" (getf job :local))))
      (when (getf job :source)
        (unless (equalp (getf job :bytes) (read-bytes (getf job :source)))
          (error "Sorgente cambiata prima della conservazione: ~A" (getf job :source)))))
    (ensure-directories-exist parent)
    (unless append-p (sb-posix:mkdir destination #o700))
    (dolist (job (reverse *jobs*))
      (let ((path (merge-pathnames (getf job :local) destination)))
        (unless (eq (getf job :operation) :rename)
          (ensure-directories-exist path)
          (with-open-file (stream path :direction :output :element-type '(unsigned-byte 8)
                                      :if-exists :error :if-does-not-exist :create)
            (write-sequence (getf job :bytes) stream))
          (unless (equalp (getf job :bytes) (read-bytes path)) (error "Copia incoerente: ~A" path))
          (when (getf job :data)
            (unless (equalp (getf job :data) (read-data path)) (error "Dato copiato incoerente: ~A" path))))
        (incf total (length (getf job :bytes)))))
    (when append-p (publish-appended-catalog destination))
    (format t "Conservati ~D file, ~D byte, ~D entry. Cartella: ~A~%"
            (length *jobs*) total (length *entries*) destination)))
(defun main ()
  "CONFIG è un dato single-form, mai LOAD/EVAL. Nessuna esecuzione di campagne."
  (unless (= 2 (length sb-ext:*posix-argv*)) (error "Fornire un solo file CONFIG.lisp."))
  (let* ((config (schema-one (read-data (second sb-ext:*posix-argv*))))
         (*root* (truename (uiop:ensure-directory-pathname (getf config :repo-root))))
         (mode (or (getf config :mode) :create))
         (catalog-path (merge-pathnames "catalogo.lisp"
                                        (merge-pathnames (getf config :path-base) *root*)))
         (previous-bytes (and (eq mode :append) (read-bytes catalog-path)))
         (previous (and previous-bytes (schema-one (read-byte-datum previous-bytes))))
         (*jobs* nil) (*entries* (and previous (reverse (copy-list (getf previous :entries)))))
         (*raw-entries* (and previous (reverse (copy-list (getf previous :associated-raw-files)))))
         (*reports* nil))
    (unless (member mode '(:create :append)) (error "Mode invalido: ~S" mode))
    (when previous
      (unless (eq (getf previous :kind) :evidence-catalog) (error "Catalogo precedente inatteso."))
      (dolist (old (getf previous :entries))
        (when (eq (getf old :artifact-kind) :command-verification)
          (push (cons (getf old :artifact)
                      (read-data (merge-pathnames (getf old :artifact)
                                                 (uiop:pathname-directory-pathname catalog-path)))) *reports*)))
      (let ((history (getf config :catalog-history-artifact)))
        (unless history (error "Append richiede catalog-history-artifact nuovo."))
        (push-job history previous-bytes :source (namestring catalog-path)
                  :data previous :operation :rename)
        (entry :prior-catalog history :artifact-schema-version (schema-value previous)
               :artifact-kind (getf previous :kind))))
    (dolist (item (getf config :process-reports)) (collect-process item))
    (dolist (item (getf config :coverage-attempts)) (collect-coverage item))
    (dolist (item (getf config :mutation-attempts)) (collect-mutation item))
    (dolist (item (getf config :reviews)) (collect-review item))
    (dolist (item (getf config :data-artifacts)) (collect-source-datum item))
    (dolist (item (getf config :raw-files)) (collect-raw-file item))
    (let ((catalog (list :schema-version 1 :kind :evidence-catalog :date (getf config :date)
                     :path-base (getf config :path-base) :formats nil
                     :metadata-policy :read-from-artifact-without-inference
                     :entries (reverse *entries*) :associated-raw-files (reverse *raw-entries*)
                     :pending (getf config :pending)
                     :limits '(:runtime-only :local-agent-readings-not-human-approval
                               :no-engine-or-release-qualification))))
      (if previous
          (push-job (or (getf config :catalog-stage-artifact) "catalogo-stage.lisp")
                    (data-bytes catalog) :data catalog :operation :catalog-stage)
          (datum-job "catalogo.lisp" catalog)))
    (commit-plan config)))
(main)
