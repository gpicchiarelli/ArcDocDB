(defun save-data (data name)
  (with-open-file (stream (merge-pathnames name *base*) :direction :output
                           :if-exists :error :if-does-not-exist :create :external-format :utf-8)
    (let ((*print-readably* t)) (write data :stream stream :pretty t) (terpri stream))))
(defun same-bytes-p (left right)
  (with-open-file (a left :element-type '(unsigned-byte 8))
    (with-open-file (b right :element-type '(unsigned-byte 8))
      (unless (= (file-length a) (file-length b)) (return-from same-bytes-p nil))
      (let ((x (make-array 65536 :element-type '(unsigned-byte 8)))
            (y (make-array 65536 :element-type '(unsigned-byte 8))))
        (loop for nx = (read-sequence x a) for ny = (read-sequence y b)
              do (unless (and (= nx ny) (equalp (subseq x 0 nx) (subseq y 0 ny)))
                   (return nil))
              when (zerop nx) return t)))))
(defun git-blob (path)
  (string-right-trim '(#\Newline #\Return)
    (uiop:run-program (list "git" "hash-object" (namestring path)) :output :string)))
(defun copy-raw (source name role &optional parent)
  (let ((target (merge-pathnames name *base*)))
    (ensure-directories-exist target)
    (if (probe-file target)
        (unless (same-bytes-p source target) (error "Copia esistente differente: ~A" target))
        (uiop:copy-file source target))
    (unless (same-bytes-p source target) (error "Bytes copiati differenti: ~A" target))
    (push (list :file name :role role :source (namestring (truename source))
                :bytes (arcdocdb.evidence:file-bytes target)
                :sha256 (arcdocdb.evidence:file-sha256 target)
                :git-blob (git-blob target) :byte-comparison :identical
                :parent-process-artifact parent) *files*)
    target))
(defun source-form (path)
  (with-open-file (stream path :external-format :utf-8)
    (let ((*read-eval* nil)) (read stream))))
(defun copy-record (source name role &optional parent)
  (let* ((original (arcdocdb.evidence:read-evidence source))
         (target (copy-raw source name role parent))
         (descriptor (source-form source)))
    (when (eq :compressed-evidence (getf descriptor :kind))
      (let ((payload (getf descriptor :payload)))
        (copy-raw (merge-pathnames payload source) payload :compressed-payload name)
        (setf (getf (first *files*) :parent-payload-path)
              (namestring (truename (merge-pathnames payload source))))))
    (unless (equalp original (arcdocdb.evidence:read-evidence target))
      (error "Record risolti differenti: ~A" name))
    (arcdocdb.evidence:call-with-evidence-bytes source
      (lambda (plain-source)
        (arcdocdb.evidence:call-with-evidence-bytes target
          (lambda (plain-copy)
            (unless (same-bytes-p plain-source plain-copy)
              (error "Payload espansi diversi: ~A" name))))))
    (when (eql 1 (or (getf original :schema-version) (getf original :schema)))
      (push (list :artifact name :kind role :source (namestring (truename source))
                  :status (getf original :status) :process-artifact parent) *records*))
    original))
(defun require-command (source name)
  (let ((data (copy-record source name :command-verification)))
    (unless (and (eq :ok (getf data :status)) (eql 0 (getf data :exit-code))
                 (eq :stable (getf data :source-consistency))
                 (equalp (getf data :source-blobs-before) (getf data :source-blobs-after)))
      (error "Comando non concluso/stabile: ~A" source))
    data))
