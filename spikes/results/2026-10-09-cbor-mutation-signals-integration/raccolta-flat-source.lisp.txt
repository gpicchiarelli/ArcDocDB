(defun bytes (file)
  (with-open-file (in file :element-type '(unsigned-byte 8))
    (let ((data (make-array (file-length in) :element-type '(unsigned-byte 8))))
      (unless (= (length data) (read-sequence data in)) (error "Incomplete file: ~A" file)) data)))
(defun blob (file)
  (string-right-trim '(#\Newline #\Return)
    (uiop:run-program (list "git" "hash-object" "--" (namestring file)) :output :string)))
(defun proof (file) (arcdocdb.evidence:read-evidence file))
(defun write-data (leaf data)
  (with-open-file (out (merge-pathnames leaf *base*) :direction :output :if-exists :error
                       :if-does-not-exist :create :external-format :utf-8)
    (let ((*print-readably* t)) (write data :stream out :pretty t) (terpri out))))
(defun copy-native (source leaf)
  (let ((target (merge-pathnames leaf *base*)))
    (when (probe-file target) (error "Existing destination: ~A" target))
    (uiop:copy-file source target)
    (unless (and (equalp (bytes source) (bytes target)) (equal (blob source) (blob target)))
      (error "Native byte mismatch: ~A" leaf))
    (push (list :file leaf :original-path (namestring (truename source))
                :byte-count (arcdocdb.evidence:file-bytes target) :git-blob (blob target)
                :sha256 (arcdocdb.evidence:file-sha256 target)) *native*)
    target))
(defun native-record (source leaf)
  (let* ((data (proof source)) (target (copy-native source leaf))
         (physical (with-open-file (in source :external-format :utf-8)
                     (let ((*read-eval* nil)) (read in)))))
    (when (eq :compressed-evidence (getf physical :kind))
      (copy-native (merge-pathnames (getf physical :payload) source) (getf physical :payload)))
    (unless (equalp data (proof target)) (error "Native report mismatch: ~A" leaf))
    (arcdocdb.evidence:call-with-evidence-bytes source
      (lambda (original)
        (arcdocdb.evidence:call-with-evidence-bytes target
          (lambda (copy)
            (unless (equalp (bytes original) (bytes copy)) (error "Expanded mismatch: ~A" leaf))))))
    data))
(defun raw-file (file base)
  (let* ((octets (bytes file)) (text (sb-ext:octets-to-string octets :external-format :utf-8)))
    (unless (equalp octets (sb-ext:string-to-octets text :external-format :utf-8))
      (error "UTF8 roundtrip mismatch: ~A" file))
    (incf *raw-count*)
    (list :path (enough-namestring file base) :original-path (namestring (truename file))
          :byte-count (length octets) :git-blob (blob file) :content text)))
(defun wrapper (directory leaf kind)
  (let* ((base (truename directory))
         (files (sort (remove-if-not
                       (lambda (file) (or (string= "report.lisp" (file-namestring file))
                                          (equal "log" (pathname-type file))
                                          (search "isolated-build.lisp" (namestring file))))
                       (directory (merge-pathnames "**/*.*" base))) #'string< :key #'namestring))
         (reports (loop for file in files when (string= "report.lisp" (file-namestring file))
                        collect (list :path (enough-namestring file base) :original-report (proof file))))
         (data (list :schema-version 1 :kind kind :original-directory (namestring base)
                     :original-reports reports :raw-files (mapcar (lambda (file) (raw-file file base)) files)
                     :limits '(:original-bytes-and-statuses-preserved :no-gate-promotion))))
    (unless reports (error "No original report: ~A" directory))
    (write-data leaf data)
    (let ((saved (proof (merge-pathnames leaf *base*))))
      (dolist (entry (getf saved :raw-files))
        (let ((encoded (sb-ext:string-to-octets (getf entry :content) :external-format :utf-8)))
          (unless (and (= (length encoded) (getf entry :byte-count))
                       (equalp encoded (bytes (getf entry :original-path)))
                       (equal (getf entry :git-blob) (blob (getf entry :original-path))))
            (error "Saved wrapper bytes mismatch: ~A" leaf))))
      (unless (equalp saved data) (error "Saved wrapper report mismatch: ~A" leaf)))
    (push (list :artifact leaf :raw-files (length files) :reports (length reports)
                :git-blob (blob (merge-pathnames leaf *base*))) *wrapper-observations*)
    (format t "Preserved ~A: ~D original files.~%" leaf (length files))))
