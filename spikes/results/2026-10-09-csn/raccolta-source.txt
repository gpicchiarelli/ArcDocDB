(require :asdf)
(require :sb-posix)
(load "tools/evidence-storage.lisp")
(defun save-data (data path)
  (with-open-file (stream path :direction :output :if-exists :supersede)
    (let ((*print-readably* t)) (write data :stream stream :pretty t) (terpri stream))))
(defun copy-exact (source target)
  (ensure-directories-exist target)
  (if (probe-file target)
      (unless (equal (arcdocdb.evidence:file-sha256 source)
                     (arcdocdb.evidence:file-sha256 target))
        (error "Target diverso, nessuna sovrascrittura: ~A" target))
      (uiop:copy-file source target))
  (unless (equal (arcdocdb.evidence:file-sha256 source)
                 (arcdocdb.evidence:file-sha256 target))
    (error "Copia non integra: ~A" target))
  target)
(defun descriptor (path)
  (with-open-file (stream path)
    (let ((*read-eval* nil)) (read stream))))
(defun copy-record (source target)
  (let ((data (descriptor source)))
    (if (eq (getf data :kind) :compressed-evidence)
        (let* ((payload (merge-pathnames (getf data :payload) source))
               (new-name (concatenate 'string (file-namestring target) ".gz"))
               (new-payload (merge-pathnames new-name target)))
          (copy-exact payload new-payload)
          (setf (getf data :payload) new-name)
          (when (probe-file target)
            (unless (equal (descriptor target) data) (error "Descriptor diverso: ~A" target)))
          (save-data data target))
        (copy-exact source target)))
  (let ((original (arcdocdb.evidence:read-evidence source))
        (copy (arcdocdb.evidence:read-evidence target)))
    (unless (equalp original copy) (error "Record risolti diversi: ~A" target)))
  target)
(defun raw-entry (source target)
  (copy-exact source target)
  (list :file (file-namestring target) :bytes (arcdocdb.evidence:file-bytes target)
        :sha256 (arcdocdb.evidence:file-sha256 target) :source (namestring source)))
(defun publish ()
  (let* ((base (merge-pathnames "spikes/results/2026-10-09-csn/" (truename "./")))
         (catalog (merge-pathnames "catalogo.lisp" base)) (entries nil) (raw nil))
    (ensure-directories-exist catalog)
    (dolist (directory (sort (uiop:subdirectories "spikes/out/") #'string< :key #'namestring))
      (let ((name (car (last (pathname-directory directory)))))
        (when (search "-command-" name)
          (dolist (leaf '("report.lisp" "conservazione.lisp"))
            (let ((source (merge-pathnames leaf directory)))
              (when (probe-file source)
                (let* ((target-name (format nil "~A-~A" name leaf))
                       (target (merge-pathnames target-name base))
                       (data (arcdocdb.evidence:read-evidence source)))
                  (unless (eql 1 (getf data :schema-version)) (error "Schema assente: ~A" source))
                  (unless (eq (getf data :status) :running)
                    (copy-record source target)
                    (push (list :artifact target-name :kind (getf data :kind)
                                :status (getf data :status) :source (namestring source)
                                :sha256 (arcdocdb.evidence:file-sha256 target)) entries)))))))))
    (labels ((collect-raw (directory prefix)
               (dolist (source (uiop:directory-files directory))
                 (unless (equal (pathname-type source) "fasl")
                   (push (raw-entry source (merge-pathnames
                                           (concatenate 'string prefix (file-namestring source)) base)) raw)))
               (dolist (subdirectory (uiop:subdirectories directory))
                 (unless (string= (car (last (pathname-directory subdirectory))) "fasl")
                   (collect-raw subdirectory
                                (format nil "~A~A_" prefix (car (last (pathname-directory subdirectory)))))))))
      (dolist (directory (uiop:subdirectories "spikes/out/"))
        (let ((scope (car (last (pathname-directory directory)))))
          (when (or (uiop:string-prefix-p "csn-coverage" scope)
                    (uiop:string-prefix-p "csn-mutations" scope))
            (collect-raw directory (concatenate 'string scope "_"))))))
    (when (probe-file "/tmp/arcdocdb-csn-review.lisp")
      (let ((target (merge-pathnames "revisione-indipendente.lisp" base)))
        (copy-record "/tmp/arcdocdb-csn-review.lisp" target)
        (push (list :artifact (file-namestring target) :kind :c1-review) entries)))
    (push (raw-entry *load-truename* (merge-pathnames "raccolta-source.txt" base)) raw)
    (dolist (leaf '("inspect-csn-results.lisp" "inspect-csn-results-failed-source.txt"
                   "inspect-csn-results-second-failed-source.txt"))
      (let ((source (merge-pathnames leaf #P"/tmp/")))
        (when (probe-file source)
          (push (raw-entry source (merge-pathnames (concatenate 'string leaf ".txt") base)) raw))))
    (unless entries (error "Nessun registro concluso."))
    (save-data (list :schema-version 1 :kind :evidence-catalog :scope :csn-registry
                     :entries (nreverse entries) :associated-raw-files (nreverse raw)
                     :limits '(:local-checks :no-engine-qualification :raw-output-embedded-once)) catalog)
    (write (list :schema-version 1 :kind :evidence-publication :status :ok
                 :catalog (namestring catalog) :records (length entries) :raw-files (length raw)))
    (terpri)))
(defun self-test ()
  (let* ((base (merge-pathnames (format nil "spikes/out/collector-fixture-~D/" (sb-posix:getpid))
                                (truename "./")))
         (source (merge-pathnames "source.lisp" base)) (payload (merge-pathnames "source.lisp.gz" base))
         (target (merge-pathnames "copy.lisp" base)))
    (ensure-directories-exist source)
    (save-data '(:schema-version 1 :kind :fixture :status :ok :value "integro") source)
    (copy-record source target)
    (copy-record source target)
    (uiop:run-program (list "gzip" "-n" "-9" "-c" (namestring source)) :output payload)
    (save-data (list :schema-version 1 :kind :compressed-evidence :codec :gzip
                     :payload (file-namestring payload)
                     :uncompressed-bytes (arcdocdb.evidence:file-bytes source)
                     :uncompressed-sha256 (arcdocdb.evidence:file-sha256 source)
                     :compressed-bytes (arcdocdb.evidence:file-bytes payload)
                     :compressed-sha256 (arcdocdb.evidence:file-sha256 payload))
               (merge-pathnames "compressed.lisp" base))
    (copy-record (merge-pathnames "compressed.lisp" base) (merge-pathnames "renamed.lisp" base))
    (save-data '(:schema-version 1 :kind :fixture :status :changed) target)
    (let ((rejected nil))
      (handler-case (copy-record source target) (error () (setf rejected t)))
      (unless rejected (error "Sovrascrittura diversa accettata.")))
    (write '(:schema-version 1 :kind :collector-self-test :status :ok :checks 4)) (terpri)))
(let ((args (uiop:command-line-arguments)))
  (cond ((equal args '("--self-test")) (self-test))
        ((equal args '("--publish")) (publish))
        (t (error "Usare --self-test o --publish."))))
