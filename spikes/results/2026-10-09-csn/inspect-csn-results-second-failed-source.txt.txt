(require :asdf)
(load "tools/evidence-storage.lisp")
(let* ((r (arcdocdb.evidence:read-evidence "spikes/out/4000520949-command-4114-0/report.lisp"))
       (b (let ((*read-eval* nil)) (read-from-string (getf r :stdout)))))
  (unless (and (eq (getf r :status) :ok) (eq (getf r :source-consistency) :stable)
               (eq (getf b :status) :ok) (eq (getf b :source-consistency) :stable))
    (error "Registro benchmark non valido."))
  (let ((samples 0) (attempts 0) (serial nil) (parallel nil))
    (dolist (c (getf b :serial))
      (let ((rates nil))
        (dolist (s (getf c :samples))
          (incf samples)
          (incf attempts (length (getf s :attempts)))
          (dolist (a (getf s :attempts))
            (unless (zerop (getf a :heap-bytes)) (error "Allocazione CSN misurata.")))
          (let ((m (getf s :measurement)))
            (unless (and (eq (getf m :status) :ok) (>= (getf m :raw-ticks) 20)
                         (plusp (getf m :seconds))) (error "Finestra CSN invalida."))
            (push (getf m :pairs-per-second) rates)))
        (push (list :scenario (getf c :scenario) :capacity (getf c :capacity)
                    :initial-high (getf c :initial-high) :initial-low (getf c :initial-low)
                    :median-pairs-per-second (third (sort rates #'<))) serial)))
    (unless (= samples 50) (error "Numero di campioni CSN inatteso."))
    (unless (= (length (getf b :parallel)) 18) (error "Repliche parallele mancanti."))
    (dolist (mode '(:shared-archive :archive-per-worker))
      (dolist (n '(1 2 4))
        (let* ((cs (remove-if-not (lambda (c) (and (eq mode (getf c :mode))
                                                  (= n (getf c :worker-count))))
                                  (getf b :parallel)))
               (rates (sort (mapcar (lambda (c) (getf c :pairs-per-second)) cs) #'<)))
          (unless (and (= (length cs) 3) (every (lambda (c) (eq (getf c :status) :ok)) cs))
            (error "Campagna parallela incompleta."))
          (push (list :mode mode :workers n :median-pairs-per-second (second rates)
                      :busy-retries (reduce #'+ cs :key (lambda (c) (getf c :busy-retries)))
                      :full-retries (reduce #'+ cs :key (lambda (c) (getf c :full-retries)))) parallel))))
    (write (list :schema-version 1 :kind :csn-results-audit :status :ok
                 :serial-samples samples :measured-attempts attempts :serial-heap-bytes 0
                 :serial (nreverse serial) :parallel-replicas 18 :parallel (nreverse parallel)
                 :source "spikes/out/4000520949-command-4114-0/report.lisp"
                 :limits '(:local-microbenchmark :oracle-overhead :external-load-uncontrolled)) :pretty t)
    (terpri)))
