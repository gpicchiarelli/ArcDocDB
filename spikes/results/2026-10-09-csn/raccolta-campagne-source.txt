(require :asdf)
(load "/tmp/collect-csn-evidence.lisp")
(let* ((base (merge-pathnames "spikes/results/2026-10-09-csn/" (truename "./")))
       (catalog (merge-pathnames "catalogo.lisp" base))
       (data (arcdocdb.evidence:read-evidence catalog)))
  (dolist (directory (uiop:subdirectories "spikes/out/"))
    (let ((name (car (last (pathname-directory directory)))))
      (when (search "-check-" name)
        (dolist (leaf '("report.lisp" "conservazione.lisp"))
          (let ((source (merge-pathnames leaf directory)))
            (when (probe-file source)
              (let ((target-name (format nil "campagna-~A-~A" name leaf)))
                (copy-record source (merge-pathnames target-name base))
                (pushnew (list :artifact target-name :kind :spike-campaign
                               :source (namestring source)) (getf data :entries) :test #'equal))))))))
  (pushnew (raw-entry *load-truename* (merge-pathnames "raccolta-campagne-source.txt" base))
           (getf data :associated-raw-files) :test #'equal)
  (save-data data catalog)
  (write (list :schema-version 1 :kind :campaign-publication :status :ok
               :records (length (getf data :entries))))
  (terpri))
