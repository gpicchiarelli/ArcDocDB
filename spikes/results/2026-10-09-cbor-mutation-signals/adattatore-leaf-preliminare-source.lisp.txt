(require :asdf)
(load "tools/evidence-storage.lisp")
(defparameter *base* #p"spikes/results/2026-10-09-cbor-mutation-signals/")
(defun path (name) (merge-pathnames name *base*))
(defun proof (name) (arcdocdb.evidence:read-evidence (path name)))
(defun save-proof (name data)
  (with-open-file (out (path name) :direction :output :if-exists :supersede
                       :if-does-not-exist :create :external-format :utf-8)
    (let ((*print-readably* t)) (write data :stream out :pretty t) (terpri out))))
(defun blob (file)
  (string-right-trim '(#\Newline #\Return)
    (uiop:run-program (list "git" "hash-object" (namestring file)) :output :string)))
(defun fingerprint (file)
  (list :bytes (arcdocdb.evidence:file-bytes file)
        :sha256 (arcdocdb.evidence:file-sha256 file) :git-blob (blob file)))
(defun copy-exact (source target)
  (when (probe-file target) (error "Target already present: ~A" target))
  (uiop:copy-file source target)
  (unless (equal (fingerprint source) (fingerprint target))
    (error "Copy differs: ~A" target)))
(defun remap (data moves)
  (cond ((stringp data) (or (second (assoc data moves :test #'string=)) data))
        ((consp data) (cons (remap (car data) moves) (remap (cdr data) moves)))
        (t data)))
