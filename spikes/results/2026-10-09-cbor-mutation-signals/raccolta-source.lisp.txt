(require :asdf)
(require :sb-posix)
(load "tools/evidence-storage.lisp")
(defparameter *destination* #p"spikes/results/2026-10-09-cbor-mutation-signals/")
(defun bytes-of (path)
  (with-open-file (in path :element-type '(unsigned-byte 8))
    (let ((data (make-array (file-length in) :element-type '(unsigned-byte 8))))
      (unless (= (read-sequence data in) (length data)) (error "Incomplete file ~A" path)) data)))
(defun text-of (path) (sb-ext:octets-to-string (bytes-of path) :external-format :utf-8))
(defun write-datum (path value)
  (ensure-directories-exist path)
  (with-open-file (out path :direction :output :if-exists :error :external-format :utf-8)
    (let ((*print-readably* t) (*print-pretty* t)) (write value :stream out) (terpri out))))
(defun evidence-datum (path)
  (arcdocdb.evidence:read-evidence path))
(defun raw-file (path base)
  (let ((raw (bytes-of path)))
    (list :path (enough-namestring path base) :original-path (namestring (truename path))
          :byte-count (length raw)
          :git-blob (string-trim '(#\Newline #\Return)
                       (uiop:run-program (list "git" "hash-object" "--" (namestring path)) :output :string))
          :content (sb-ext:octets-to-string raw :external-format :utf-8))))
(defun native-copy (source relative)
  (let ((target (merge-pathnames relative *destination*)))
    (when (probe-file target) (error "Target exists ~A" target))
    (ensure-directories-exist target)
    (uiop:copy-file source target)
    ;; Descriptor bytes and native payload filename stay exactly as originally emitted.
    (let* ((*read-eval* nil)
           (physical (with-open-file (in source :external-format :utf-8) (read in))))
      (when (eq (getf physical :kind) :compressed-evidence)
        (let ((payload (getf physical :payload)))
          (uiop:copy-file (merge-pathnames payload (uiop:pathname-directory-pathname source))
                          (merge-pathnames payload (uiop:pathname-directory-pathname target))))))
    (unless (equalp (bytes-of source) (bytes-of target)) (error "Byte mismatch ~A" source))
    (evidence-datum target)))
(defun preserve-directory (base leaf kind)
  (let* ((paths (sort (remove-if-not
                      (lambda (path)
                        (or (string= (file-namestring path) "report.lisp")
                            (string= (or (pathname-type path) "") "log")
                            (search "isolated-build.lisp" (namestring path))))
                      (directory (merge-pathnames "**/*.*" base)))
                     #'string< :key #'namestring))
         (reports (loop for path in paths when (string= (file-namestring path) "report.lisp")
                        collect (list :path (enough-namestring path base) :original-report (evidence-datum path))))
         (data (list :schema-version 1 :kind kind :scope :cbor-mutation-processes
                     :original-directory (namestring (truename base)) :original-reports reports
                     :raw-files (mapcar (lambda (path) (raw-file path base)) paths)
                     :limits '(:original-bytes-and-statuses-preserved :no-status-promotion
                               :tool-verification-only :not-c1-or-mcdc))))
    (unless reports (error "No reports in ~A" base))
    (write-datum (merge-pathnames leaf *destination*) data)
    (format t "Preserved ~A (~D raw files)~%" leaf (length paths))))
(defun main ()
  (let ((commands (sort (directory "spikes/out/*command*/report.lisp") #'string< :key #'namestring))
        (fixtures (sort (append (directory "spikes/out/*cbor-header-process-self-test*/report.lisp")
                                (directory "spikes/out/*cbor-structure-worker-self-test*/report.lisp"))
                        #'string< :key #'namestring)))
    (loop for path in commands for index from 0
          do (let ((prefix (format nil "commands/~D/" index)))
               (native-copy path (concatenate 'string prefix "report.lisp"))
               (let ((preservation (merge-pathnames "conservazione.lisp" (uiop:pathname-directory-pathname path))))
                 (when (probe-file preservation)
                   (native-copy preservation (concatenate 'string prefix "conservazione.lisp"))))))
    (loop for path in fixtures for index from 0
          do (preserve-directory (uiop:pathname-directory-pathname path)
                                  (format nil "fixture-~D.lisp" index) :process-fixture-output))
    (preserve-directory #p"spikes/out/cbor-header-signals/" "header-mutazioni.lisp" :mutation-output)
    (preserve-directory #p"spikes/out/cbor-structure-signals/" "struttura-mutazioni.lisp" :mutation-output)
    (let ((spike-reports (sort (directory "spikes/out/*-check-*/report.lisp") #'string< :key #'namestring)))
      (loop for path in spike-reports for index from 0
            do (let ((base (uiop:pathname-directory-pathname path))
                     (prefix (format nil "spikes/~D/" index)))
                 (dolist (report (sort (directory (merge-pathnames "*.lisp" base)) #'string< :key #'namestring))
                   (native-copy report (concatenate 'string prefix (file-namestring report))))))))
  (format t "Preservation finished; catalog publication remains explicit.~%"))
(main)
