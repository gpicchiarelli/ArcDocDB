(require :asdf)
(require :sb-posix)
(defpackage #:arcdocdb.ci (:use #:cl))
(in-package #:arcdocdb.ci)
(defparameter *ci-root* (truename "./"))
(defparameter *ci-entry* (merge-pathnames "tools/ci.lisp" *ci-root*))
(defparameter *ci-sbcl* (namestring sb-ext:*runtime-pathname*))
(load (merge-pathnames "tools/evidence-storage.lisp" *ci-root*))
(load (merge-pathnames "tests/ci-catalog.lisp" *ci-root*))
(dolist (module '("data" "identity" "support" "tests" "coverage" "measure" "archive" "producer"))
  (load (merge-pathnames (format nil "tools/ci/~A.lisp" module) *ci-root*)))
(let ((before 0) (after 0) (cases nil))
  (dolist (path (directory "spikes/out/evidence-cleanup/ci-input/*/manifest.lisp"))
    (let* ((old (uiop:pathname-directory-pathname (truename path)))
           (manifest (ci-verify-packet old)) (job (getf manifest :job))
           (new (ci-mkdir (format nil "spikes/out/evidence-cleanup/repack/~A/" job)))
           (work (ci-mkdir (merge-pathnames "work/" new))))
      (dolist (row (getf manifest :files))
        (let ((name (getf row :path)))
          (when (and (uiop:string-prefix-p "raw/" name) (not (equal (pathname-type name) "gz")))
            (arcdocdb.evidence:call-with-evidence-bytes
             (merge-pathnames name old)
             (lambda (plain) (ci-copy-new (truename plain) (merge-pathnames (subseq name 4) work)))))))
      (let* ((packet (ci-pack-job new work job (getf manifest :identity) (ci-read (merge-pathnames "result.lisp" old))))
             (old-size (loop for file in (ci-files old) sum (ci-bytes file)))
             (new-size (loop for file in (ci-files packet) sum (ci-bytes file))))
        (incf before old-size) (incf after new-size)
        (push (list :job job :before old-size :after new-size) cases))))
  (assert (< after +ci-packet-bytes+))
  (write (list :schema-version 1 :kind :ci-archive-size-comparison :status :ok
               :original-run "40010510208" :before-bytes before :after-bytes after
               :budget +ci-packet-bytes+ :raw-copy-hashes-verified t :jobs (nreverse cases)
               :limits '(:repack-only :not-a-rerun :not-a-server-attestation))) (terpri))
