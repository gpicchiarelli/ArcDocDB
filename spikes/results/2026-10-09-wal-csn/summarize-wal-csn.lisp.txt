(require :asdf)
(load "tools/evidence-storage.lisp")
(defun midpoint (numbers)
  (unless (every #'realp numbers) (error "Metrica numerica mancante."))
  (let* ((ordered (sort (copy-list numbers) #'<)) (n (length ordered)))
    (unless (plusp n) (error "Campione vuoto."))
    (if (oddp n) (nth (floor n 2) ordered)
        (/ (+ (nth (1- (/ n 2)) ordered) (nth (/ n 2) ordered)) 2))))
(defun show-summary (report)
  (write (list :schema-version 1 :kind (getf report :kind) :status (getf report :status)
               :source-consistency (getf report :source-consistency))) (terpri)
  (when (stringp (getf report :stdout))
    (dolist (line (uiop:split-string (getf report :stdout) :separator '(#\Newline)))
      (when (or (search "test " line) (search "WAL/CSN:" line) (search "CSN:" line))
        (write-line line))))
  (when (equal (getf report :mode) "--bench")
    (unless (and (eq (getf report :status) :ok) (eq (getf report :source-consistency) :stable))
      (error "Campagna benchmark non riuscita: ~A" (getf report :diagnostic)))
    (dolist (campaign (getf report :serial))
      (let ((windows (mapcar (lambda (sample) (getf sample :measurement)) (getf campaign :samples))))
        (write (list :serial-base (list (getf campaign :initial-high) (getf campaign :initial-low))
                     :samples (length windows)
                     :all-zero-heap (every (lambda (window) (zerop (getf window :heap-bytes))) windows)
                     :median-cycles-per-second (midpoint (mapcar (lambda (window) (getf window :cycles-per-second)) windows))
                     :min-ticks (reduce #'min windows :key (lambda (window) (getf window :raw-ticks)))))
        (terpri)))
    (dolist (workers '(1 2 4))
      (let* ((campaigns (remove-if-not (lambda (campaign) (= (getf campaign :worker-count) workers))
                                     (getf report :parallel)))
             (attempts (mapcar (lambda (campaign) (getf campaign :measurement)) campaigns))
             (windows (mapcar (lambda (attempt) (getf attempt :measurement)) attempts)))
        (write (list :workers workers :replicas (length attempts)
                     :median-cycles-per-second (midpoint (mapcar (lambda (window) (getf window :cycles-per-second)) windows))
                     :process-heap-bytes (mapcar (lambda (attempt) (getf attempt :process-heap-bytes)) attempts)
                     :busy-failures (mapcar (lambda (window) (getf window :busy-failures)) windows)
                     :full-failures (mapcar (lambda (window) (getf window :full-failures)) windows)
                     :retries (mapcar (lambda (window) (getf window :retries)) windows))) (terpri)))))
(dolist (path (uiop:command-line-arguments)) (show-summary (arcdocdb.evidence:read-evidence path)))
