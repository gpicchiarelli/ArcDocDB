(require :asdf)
(load "tools/evidence-storage.lisp")

       (stderr (getf report :stderr)))
  (let ((start (or (search "Unhandled" stderr) 0)))
    (format t "~A~%" (subseq stderr start (min (+ start 2800) (length stderr))))))
