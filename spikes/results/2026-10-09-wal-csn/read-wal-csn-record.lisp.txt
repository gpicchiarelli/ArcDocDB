(require :asdf)
(load "tools/evidence-storage.lisp")
(let ((report (arcdocdb.evidence:read-evidence (first (uiop:command-line-arguments)))))
  (write (list :status (getf report :status) :exit-code (getf report :exit-code)
               :source-consistency (getf report :source-consistency)))
  (terpri)
  (dolist (key '(:stdout :stderr))
    (let ((value (getf report key)))
      (when (stringp value)
        (format t "~A~%~A~%" key (subseq value (max 0 (- (length value) 9000))))))))
