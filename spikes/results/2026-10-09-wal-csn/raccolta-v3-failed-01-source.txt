(load "tools/evidence-storage.lisp")
(defun save-data (data path)
  (with-open-file (stream path :direction :output :if-exists :supersede)
    (let ((*print-readably* t)) (write data :stream stream :pretty t) (terpri stream))))
(defun copy-exact (source target)
  (ensure-directories-exist target)
  (if (probe-file target)
      (unless (equal (arcdocdb.evidence:file-sha256 source)
                     (arcdocdb.evidence:file-sha256 target))
        (error "Target diverso, nessuna sovrascrittura: ~A" target))
      (uiop:copy-file source target))
  (unless (equal (arcdocdb.evidence:file-sha256 source)
                 (arcdocdb.evidence:file-sha256 target))
    (error "Copia non integra: ~A" target))
  target)
(defun descriptor (path)
  (with-open-file (stream path)
    (let ((*read-eval* nil)) (read stream))))
(defun copy-record (source target)
  (let ((data (descriptor source)))
    (if (eq (getf data :kind) :compressed-evidence)
        (let* ((payload (merge-pathnames (getf data :payload) source))
               (new-name (concatenate 'string (file-namestring target) ".gz"))
               (new-payload (merge-pathnames new-name target)))
          (copy-exact payload new-payload)
          (setf (getf data :payload) new-name)
          (when (probe-file target)
            (unless (equal (descriptor target) data) (error "Descriptor diverso: ~A" target)))
          (save-data data target))
        (copy-exact source target)))
  (let ((original (arcdocdb.evidence:read-evidence source))
        (copy (arcdocdb.evidence:read-evidence target)))
    (unless (equalp original copy) (error "Record risolti diversi: ~A" target)))
  target)
(defun raw-entry (source target &optional (base (merge-pathnames "spikes/results/2026-10-09-wal-csn/" (truename "./"))))
  (copy-exact source target)
  (list :file (enough-namestring target base) :bytes (arcdocdb.evidence:file-bytes target)
        :sha256 (arcdocdb.evidence:file-sha256 target) :source (namestring source)))
(defun collect-raw (directory prefix base)
  (let ((raw nil))
    (dolist (source (uiop:directory-files directory))
      (unless (equal (pathname-type source) "fasl")
        (push (raw-entry source (merge-pathnames
                                (concatenate 'string prefix (file-namestring source)) base) base) raw)))
    (dolist (subdirectory (uiop:subdirectories directory))
      (unless (member (car (last (pathname-directory subdirectory)))
                      '("fasl" "src" "tests") :test #'string=)
        (setf raw (nconc raw (collect-raw subdirectory
                                        (format nil "~A~A/" prefix (car (last (pathname-directory subdirectory))))
                                        base)))))
    raw))
(defun migrate ()
  (let* ((base (merge-pathnames "spikes/results/2026-10-09-wal-csn/" (truename "./")))
         (out (merge-pathnames "spikes/out/" (truename "./")))
         (catalog (arcdocdb.evidence:read-evidence (merge-pathnames "catalogo.lisp" base)))
         (moves nil))
    ;; Validate the complete plan before moving any byte; historical Git objects remain intact.
    (dolist (entry (getf catalog :associated-raw-files))
      (let ((name (getf entry :file)))
        (when (and (not (find #\/ name))
                   (not (uiop:string-suffix-p "_mutated-source.txt" name))
                   (or (uiop:string-prefix-p "wal-csn-coverage" name)
                       (uiop:string-prefix-p "wal-csn-mutations" name)))
          (let* ((old (merge-pathnames name base))
                 (relative (enough-namestring (getf entry :source) out))
                 (new (merge-pathnames relative base)))
            (unless (and (uiop:relative-pathname-p relative)
                         (or (uiop:string-prefix-p "wal-csn-coverage" relative)
                             (uiop:string-prefix-p "wal-csn-mutations" relative))
                         (= (getf entry :bytes) (arcdocdb.evidence:file-bytes old))
                         (equal (getf entry :sha256) (arcdocdb.evidence:file-sha256 old))
                         (not (probe-file new)))
              (error "Piano di spostamento invalido: ~A" name))
            (push (list old new (getf entry :sha256)) moves)))))
    (dolist (move moves)
      (destructuring-bind (old new sha256) move
        (ensure-directories-exist new)
        (rename-file old new)
        (unless (equal sha256 (arcdocdb.evidence:file-sha256 new))
          (error "Impronta diversa dopo spostamento: ~A" new))))
    (write (list :schema-version 1 :kind :raw-evidence-layout :status :ok
                 :moved-files (length moves) :payload-bytes-changed 0)) (terpri)))
(defun catalog-counts (path)
  (let ((data (arcdocdb.evidence:read-evidence path)))
    (unless (eq (getf data :kind) :evidence-catalog) (error "Catalogo mancante."))
    (values (length (getf data :entries)) (length (getf data :associated-raw-files)))))
(defun publish ()
  (let* ((base (merge-pathnames "spikes/results/2026-10-09-wal-csn/" (truename "./")))
         (catalog (merge-pathnames "catalogo.lisp" base)) (entries nil) (raw nil))
    (ensure-directories-exist catalog)
    (dolist (directory (sort (uiop:subdirectories "spikes/out/") #'string< :key #'namestring))
      (let ((name (car (last (pathname-directory directory)))))
        (when (or (search "-command-" name) (search "-check-" name)
                  (uiop:string-prefix-p "wal-csn-bench" name)
                  (uiop:string-prefix-p "wal-csn-self-test" name))
          (dolist (leaf '("report.lisp" "conservazione.lisp" "self-test-metrics.lisp"))
            (let ((source (merge-pathnames leaf directory)))
              (when (probe-file source)
                (let* ((target-name (format nil "~A-~A" name leaf))
                       (target (merge-pathnames target-name base))
                       (data (arcdocdb.evidence:read-evidence source)))
                  (unless (eql 1 (getf data :schema-version)) (error "Schema assente: ~A" source))
                  (unless (eq (getf data :status) :running)
                    (copy-record source target)
                    (push (list :artifact target-name :kind (getf data :kind)
                                :status (getf data :status) :source (namestring source)
                                :sha256 (arcdocdb.evidence:file-sha256 target)) entries)))))))))
    (dolist (directory (uiop:subdirectories "spikes/out/"))
      (let ((scope (car (last (pathname-directory directory)))))
        (when (or (uiop:string-prefix-p "wal-csn-coverage" scope)
                  (uiop:string-prefix-p "wal-csn-mutations" scope))
          (setf raw (nconc raw (collect-raw directory (concatenate 'string scope "/") base))))))
    (dolist (directory (uiop:subdirectories "spikes/out/"))
      (let ((name (car (last (pathname-directory directory)))))
        (when (uiop:string-prefix-p "wal-csn-mutations" name)
          (let ((report (merge-pathnames "report.lisp" directory)))
            (when (probe-file report)
              (dolist (target (getf (arcdocdb.evidence:read-evidence report) :targets))
                (let* ((scope (getf target :name))
                       (source (merge-pathnames (getf target :file)
                                                (merge-pathnames (format nil "~A/" scope) directory))))
                  (when (probe-file source)
                    (push (raw-entry source (merge-pathnames
                                             (format nil "~A_~A_mutated-source.txt" name scope) base)) raw)))))))))
    (push (raw-entry *load-truename* (merge-pathnames "raccolta-v3-source.txt" base)) raw)
    (push (raw-entry #p"/tmp/collect-wal-csn-evidence-v2.lisp"
                     (merge-pathnames "raccolta-v2-source.txt" base)) raw)
    (dolist (leaf '("summarize-wal-csn.lisp" "read-wal-csn-record.lisp" "read-wal-csn-diagnostic.lisp"))
      (let ((source (merge-pathnames leaf #p"/tmp/")))
        (when (probe-file source)
          (push (raw-entry source (merge-pathnames (concatenate 'string leaf ".txt") base)) raw))))
    (push (raw-entry #p"/tmp/collect-wal-csn-evidence.lisp"
                     (merge-pathnames "raccolta-source.txt" base)) raw)
    (unless entries (error "Nessun registro concluso."))
    (save-data (list :schema-version 1 :kind :evidence-catalog :scope :wal-csn-bridge
                     :entries (nreverse entries) :associated-raw-files (nreverse raw)
                     :limits '(:local-checks :no-engine-qualification :raw-output-embedded-once)) catalog)
    (write (list :schema-version 1 :kind :evidence-publication :status :ok
                 :catalog (namestring catalog) :records (nth-value 0 (catalog-counts catalog))
                 :raw-files (nth-value 1 (catalog-counts catalog))))
    (terpri)))
(defun self-test ()
  (let* ((base (merge-pathnames (format nil "spikes/out/collector-fixture-~D/" (sb-posix:getpid))
                                (truename "./")))
         (source (merge-pathnames "source.lisp" base)) (payload (merge-pathnames "source.lisp.gz" base))
         (target (merge-pathnames "copy.lisp" base)))
    (ensure-directories-exist source)
    (save-data '(:schema-version 1 :kind :fixture :status :ok :value "integro") source)
    (copy-record source target)
    (copy-record source target)
    (uiop:run-program (list "gzip" "-n" "-9" "-c" (namestring source)) :output payload)
    (save-data (list :schema-version 1 :kind :compressed-evidence :codec :gzip
                     :payload (file-namestring payload)
                     :uncompressed-bytes (arcdocdb.evidence:file-bytes source)
                     :uncompressed-sha256 (arcdocdb.evidence:file-sha256 source)
                     :compressed-bytes (arcdocdb.evidence:file-bytes payload)
                     :compressed-sha256 (arcdocdb.evidence:file-sha256 payload))
               (merge-pathnames "compressed.lisp" base))
    (copy-record (merge-pathnames "compressed.lisp" base) (merge-pathnames "renamed.lisp" base))
    (save-data '(:schema-version 1 :kind :fixture :status :changed) target)
    (let ((rejected nil))
      (handler-case (copy-record source target) (error () (setf rejected t)))
      (unless rejected (error "Sovrascrittura diversa accettata.")))
    (let ((catalog (merge-pathnames "count-fixture.lisp" base)))
      (save-data '(:schema-version 1 :kind :evidence-catalog
                   :entries ((:artifact "a") (:artifact "b") (:artifact "c"))
                   :associated-raw-files ((:file "a") (:file "b"))) catalog)
      (unless (and (= 3 (nth-value 0 (catalog-counts catalog)))
                   (= 2 (nth-value 1 (catalog-counts catalog))))
        (error "Conteggi persistiti del catalogo errati.")))
    (let* ((original (merge-pathnames "original/" base))
           (index (merge-pathnames "cover-index.html" original))
           (page (merge-pathnames "nested/page.html" original))
           (ignored (merge-pathnames "fasl/probe.fasl" original))
           (published (merge-pathnames "published/" base)))
      (dolist (path (list index page ignored)) (ensure-directories-exist path))
      (uiop:write-string-into-file "<a href='nested/page.html'>copertura</a>" index :if-exists :supersede)
      (uiop:write-string-into-file "<p>pagina originale</p>" page :if-exists :supersede)
      (uiop:write-string-into-file "ricostruibile" ignored :if-exists :supersede)
      (let ((entries (collect-raw original "coverage/" published)))
        (unless (equal (sort (mapcar (lambda (entry) (getf entry :file)) entries) #'string<)
                       '("coverage/cover-index.html" "coverage/nested/page.html"))
          (error "Percorsi relativi/filtri non conservati."))
        (unless (and (probe-file (merge-pathnames "nested/page.html"
                                                 (merge-pathnames "coverage/cover-index.html" published)))
                     (equal (arcdocdb.evidence:file-sha256 page)
                            (arcdocdb.evidence:file-sha256 (merge-pathnames "coverage/nested/page.html" published))))
          (error "Link relativo o bytes HTML alterati."))))
    (write '(:schema-version 1 :kind :collector-self-test :status :ok :checks 7)) (terpri)))
(let ((args (uiop:command-line-arguments)))
  (cond ((equal args '("--self-test")) (self-test))
        ((equal args '("--publish")) (publish))
        ((equal args '("--migrate")) (migrate))
        (t (error "Usare --self-test, --publish o --migrate."))))
