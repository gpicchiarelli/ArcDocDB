;;;; Esegue solo guardie del collector su fixture temporanee, mai campagne del prodotto.
(require :asdf)
(require :sb-posix)
(let ((*package* (find-package :cl-user)) (*read-eval* nil))
  (with-open-file (stream "/tmp/collect-utf8-evidence.lisp" :external-format :utf-8)
    (let ((eof (gensym)))
      (loop for form = (read stream nil eof) until (eq form eof)
            unless (and (consp form) (null (cdr form)) (symbolp (car form))
                        (string= (symbol-name (car form)) "MAIN"))
              do (eval form)))))
(in-package :cl-user)
(defvar *guard-log*)
(defvar *guard-passed* nil)
(defvar *guard-failed* nil)
(defvar *reader-executed* nil)
(defun emit-guard (control &rest args)
  (apply #'format t control args))
(defun ensure-guard (value)
  (unless value (error "Asserzione della guardia fallita.")))
(defun expect-guard-error (thunk &optional (class 'error))
  (let ((caught nil))
    (handler-case (funcall thunk)
      (error (condition) (unless (typep condition class) (error condition)) (setf caught t)))
    (ensure-guard caught)))
(defun guard-case (name thunk)
  (handler-case (progn (funcall thunk) (push name *guard-passed*) (emit-guard "ok ~S~%" name))
    (error (condition)
      (push (list :name name :condition (princ-to-string condition)) *guard-failed*)
      (emit-guard "FAILED ~S: ~A~%" name condition))))
(defun write-guard-bytes (path bytes)
  (ensure-directories-exist path)
  (with-open-file (stream path :direction :output :element-type '(unsigned-byte 8)
                              :if-exists :error :if-does-not-exist :create)
    (write-sequence bytes stream)) path)
(defun write-guard-data (path data)
  (write-guard-bytes path (writer-queue.evidence::data-bytes data)))
(defun wrapper-fixture (&optional (stdout "messaggio è") (stderr ""))
  (list :schema-version 1 :kind :command-verification :status :ok :exit-code 0
        :source-consistency :stable :stdout stdout :stderr stderr))

(let* ((report-path (or (second sb-ext:*posix-argv*) "/tmp/utf8-collector-self-test.lisp"))
       (log-path (concatenate 'string (namestring (make-pathname :type nil :defaults report-path)) ".log"))
       (root (uiop:ensure-directory-pathname
              (format nil "/tmp/utf8-collector-fixture-~D-~D/" (get-universal-time) (sb-posix:getpid))))
       (*guard-passed* nil) (*guard-failed* nil))
  (writer-queue.evidence::fresh-target report-path)
  (writer-queue.evidence::fresh-target log-path)
  (sb-posix:mkdir root #o700)
  (with-open-file (*guard-log* log-path :direction :output :if-exists :error :external-format :utf-8)
    (let ((*standard-output* (make-broadcast-stream *standard-output* *guard-log*)))
    (guard-case :reader-disabled-eval
      (lambda ()
        (setf *reader-executed* nil)
        (expect-guard-error
         (lambda () (with-input-from-string (stream "#.(progn (setf cl-user::*reader-executed* t) 7)")
                      (writer-queue.evidence::read-data-stream stream))) 'reader-error)
        (ensure-guard (null *reader-executed*))))
    (guard-case :reader-extra-eof
      (lambda () (expect-guard-error
                  (lambda () (with-input-from-string (stream "(:a 1) (:b 2)")
                               (writer-queue.evidence::read-data-stream stream))) 'simple-error)))
    (guard-case :missing-and-duplicated-targets
      (lambda ()
        (let ((writer-queue.evidence::*jobs* nil))
          (expect-guard-error (lambda () (writer-queue.evidence::push-job nil #(1))) 'simple-error)
          (writer-queue.evidence::push-job "x.log" #(1))
          (expect-guard-error (lambda () (writer-queue.evidence::push-job "x.log" #(2))) 'simple-error)
          (ensure-guard (= 1 (length writer-queue.evidence::*jobs*))))))
    (guard-case :fresh-target-retains-existing-bytes
      (lambda ()
        (let ((path (write-guard-bytes (merge-pathnames "existing.log" root) #(1 0 255))))
          (expect-guard-error (lambda () (writer-queue.evidence::fresh-target path)) 'simple-error)
          (ensure-guard (equalp #(1 0 255) (writer-queue.evidence::read-bytes path))))))
    (guard-case :byte-copy-and-utf8-embedded
      (lambda ()
        (let* ((writer-queue.evidence::*root* root) (writer-queue.evidence::*jobs* nil)
               (writer-queue.evidence::*entries* nil) (writer-queue.evidence::*reports* nil)
               (source (write-guard-data (merge-pathnames "wrapper.lisp" root) (wrapper-fixture))))
          (writer-queue.evidence::collect-process (list :source source :artifact "wrapper.lisp" :role :fixture))
          (writer-queue.evidence::commit-plan '(:path-base "byte-out/"))
          (ensure-guard (equalp (writer-queue.evidence::read-bytes source)
                               (writer-queue.evidence::read-bytes (merge-pathnames "byte-out/wrapper.lisp" root))))
          (ensure-guard (string= "messaggio è" (writer-queue.evidence::read-text
                                                (merge-pathnames "byte-out/wrapper.stdout.log" root))))
          (dolist (entry writer-queue.evidence::*entries*)
            (ensure-guard (null (find #\/ (getf entry :artifact))))
            (writer-queue.evidence::schema-one
             (writer-queue.evidence::read-data
              (merge-pathnames (getf entry :artifact) (merge-pathnames "byte-out/" root))))))))
    (guard-case :nil-and-absent-channels
      (lambda ()
        (let* ((writer-queue.evidence::*root* root) (writer-queue.evidence::*jobs* nil)
               (writer-queue.evidence::*entries* nil) (writer-queue.evidence::*reports* nil)
               (data (wrapper-fixture nil nil)))
          (remf data :stderr)
          (writer-queue.evidence::collect-process
           (list :source (write-guard-data (merge-pathnames "nil-wrapper.lisp" root) data)
                 :artifact "nil-wrapper.lisp" :role :fixture))
          (ensure-guard (= 1 (length writer-queue.evidence::*jobs*)))
          (let ((absent (remove-if-not (lambda (entry) (eq (getf entry :kind) :raw-process-output-absence))
                                       writer-queue.evidence::*entries*)))
            (ensure-guard (= 2 (length absent)))
            (ensure-guard (getf (find :stdout absent :key (lambda (entry) (getf entry :channel))) :present-p))
            (ensure-guard (null (getf (find :stderr absent :key (lambda (entry) (getf entry :channel))) :present-p)))))))
    (guard-case :mutation-all-log-strings
      (lambda ()
        (let* ((writer-queue.evidence::*root* root) (writer-queue.evidence::*jobs* nil)
               (writer-queue.evidence::*entries* nil)
               (writer-queue.evidence::*reports* (list (cons "wrapper.lisp" (wrapper-fixture))))
               (base (merge-pathnames "mut-source/" root))
               (baseline (write-guard-bytes (merge-pathnames "baseline/test.log" base)
                                            (sb-ext:string-to-octets "baseline è" :external-format :utf-8)))
               (mutant (write-guard-bytes (merge-pathnames "0/test.log" base) #(65 10)))
               (extra (write-guard-bytes (merge-pathnames "extra.log" base) #(66 10))))
          (declare (ignore extra))
          (write-guard-data (merge-pathnames "report.lisp" base)
                           (list :schema-version 1 :kind :writer-queue-mutations :status :failed
                                 :baseline (list :log (namestring baseline))
                                 :mutants (list (list :name "fixture-mutant" :log (namestring mutant)))))
          (writer-queue.evidence::collect-mutation
           '(:artifact "raw.lisp" :process-artifact "wrapper.lisp" :report-source "mut-source/report.lisp"))
          (writer-queue.evidence::commit-plan '(:path-base "mut-out/"))
          (let* ((raw (writer-queue.evidence::read-data (merge-pathnames "mut-out/raw.lisp" root)))
                 (logs (getf raw :logs)))
            (ensure-guard (= 3 (length logs)))
            (dolist (log logs)
              (ensure-guard (string= (getf log :content) (writer-queue.evidence::read-text (getf log :source-path))))
              (ensure-guard (equalp (writer-queue.evidence::read-bytes (getf log :source-path))
                                   (writer-queue.evidence::read-bytes
                                    (merge-pathnames (getf log :local-file) (merge-pathnames "mut-out/" root))))))))))
    (guard-case :append-prior-byte-identity-and-repeated-name
      (lambda ()
        (let* ((writer-queue.evidence::*root* root) (writer-queue.evidence::*jobs* nil)
               (writer-queue.evidence::*entries* nil)
               (source (merge-pathnames "append-out/catalogo.lisp" root))
               (original (sb-ext:string-to-octets
                          (format nil "; byte formatting preserved~%(:schema-version 1 :kind :evidence-catalog :entries NIL)~%")
                          :external-format :utf-8))
               (previous (writer-queue.evidence::read-byte-datum original))
               (next '(:schema-version 1 :kind :evidence-catalog :entries ((:kind :prior-catalog :formats NIL :artifact "prior.lisp")))))
          (write-guard-bytes source original)
          (writer-queue.evidence::push-job "prior.lisp" original :source (namestring source) :data previous :operation :rename)
          (writer-queue.evidence::push-job "stage.lisp" (writer-queue.evidence::data-bytes next) :data next :operation :catalog-stage)
          (writer-queue.evidence::commit-plan '(:mode :append :path-base "append-out/"))
          (ensure-guard (equalp original (writer-queue.evidence::read-bytes (merge-pathnames "append-out/prior.lisp" root))))
          (ensure-guard (equalp next (writer-queue.evidence::read-data source)))
          (setf writer-queue.evidence::*jobs* nil)
          (writer-queue.evidence::push-job "prior.lisp" (writer-queue.evidence::read-bytes source)
                                          :source (namestring source) :data next :operation :rename)
          (expect-guard-error (lambda () (writer-queue.evidence::commit-plan '(:mode :append :path-base "append-out/"))) 'simple-error)
          (ensure-guard (equalp original (writer-queue.evidence::read-bytes (merge-pathnames "append-out/prior.lisp" root)))))))
    (guard-case :mismatched-catalog-data-and-bytes-before-write
      (lambda ()
        (let* ((writer-queue.evidence::*root* root) (writer-queue.evidence::*jobs* nil)
               (writer-queue.evidence::*entries* nil) (source (merge-pathnames "append-out/catalogo.lisp" root)))
          (writer-queue.evidence::push-job "bad-prior.lisp" (writer-queue.evidence::read-bytes source)
                                          :source (namestring source) :data '(:different t) :operation :rename)
          (expect-guard-error (lambda () (writer-queue.evidence::commit-plan '(:mode :append :path-base "append-out/"))) 'simple-error)
          (ensure-guard (null (probe-file (merge-pathnames "append-out/bad-prior.lisp" root)))))))
    (guard-case :catalog-top-entries-schema-one-basenames
      (lambda ()
        (let* ((writer-queue.evidence::*root* root) (writer-queue.evidence::*jobs* nil)
               (writer-queue.evidence::*entries* nil) (writer-queue.evidence::*raw-entries* nil)
               (writer-queue.evidence::*reports* nil))
          (writer-queue.evidence::collect-process
           (list :source (merge-pathnames "wrapper.lisp" root) :artifact "wrapper.lisp" :role :fixture))
          (writer-queue.evidence::datum-job "catalogo.lisp"
           (list :schema-version 1 :kind :evidence-catalog :formats nil
                 :entries (reverse writer-queue.evidence::*entries*)
                 :associated-raw-files (reverse writer-queue.evidence::*raw-entries*)))
          (writer-queue.evidence::commit-plan '(:path-base "catalog-out/"))
          (let ((catalog (writer-queue.evidence::read-data (merge-pathnames "catalog-out/catalogo.lisp" root))))
            (ensure-guard (= 1 (length (getf catalog :entries))))
            (ensure-guard (= 2 (length (getf catalog :associated-raw-files))))
            (dolist (entry (getf catalog :entries))
              (ensure-guard (null (find #\/ (getf entry :artifact))))
              (writer-queue.evidence::schema-one
               (writer-queue.evidence::read-data
                (merge-pathnames (getf entry :artifact) (merge-pathnames "catalog-out/" root)))))))))
    (emit-guard "Collector guard self-test: ~D passed, ~D failed.~%" (length *guard-passed*) (length *guard-failed*))
    (finish-output *guard-log*)))
  (write-guard-data report-path
                   (list :schema-version 1 :kind :utf8-collector-self-test :formats nil
                         :status (if *guard-failed* :failed :ok) :source "/tmp/collect-utf8-evidence.lisp"
                         :fixtures (namestring root) :recorded-at (get-universal-time)
                         :passed (reverse *guard-passed*) :failed (reverse *guard-failed*)
                         :log-source log-path :raw-stdout (writer-queue.evidence::read-text log-path)
                         :limits '(:collector-guards-only :trusted-collector-code-loaded-without-main
                                   :no-raw-data-load-or-eval :no-product-campaigns)))
  (when *guard-failed* (sb-ext:exit :code 1)))
